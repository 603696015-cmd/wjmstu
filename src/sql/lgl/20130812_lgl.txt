create table words(
  id number NOT NULL PRIMARY KEY,
  name varchar(200),
  DESCRIPTION varchar(300),
  PARENTID number,
  courseid number
)


create  SEQUENCE  words_sequence
  increment by 1 -- 每次加几个
  start with 1 -- 从1开始计数
  nomaxvalue -- 不设置最大值
  nocycle -- 一直累加，不循环
  cache 10 ;
  
  create or replace trigger auto_id_words  before insert on words for each row
when (new.id is null)
begin
     select words_sequence.nextval into :new.id from dual;
end;


alter table words add (lid number,rid number);

alter table words add (status number(1) default 0);

create table words_OP_USER
(
  userid NUMBER,
  depid  NUMBER

tablespace USERS
  pctfree 10
