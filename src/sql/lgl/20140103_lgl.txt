-- Create table 简历表
create table resume
(
  id              NUMBER not null,
  userid             NUMBER
)


-- Create sequence 
create sequence resume_sequence
minvalue 1
maxvalue 999999999999999999999999999
start with 1073
increment by 1
cache 10;

create or replace trigger auto_id_resume  before insert on resume for each row
when (new.id is null)
begin
     select resume_sequence.nextval into :new.id from dual;
end;
