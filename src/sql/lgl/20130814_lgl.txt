
//创建词汇表
create table vocabulary(
  id number NOT NULL PRIMARY KEY,
  name varchar(200),
  adduserid number,
  alteruserid number,
  addtime date,
  altertime date,
  wordid number,
  pinyin varchar(300),
  duyin varchar(300),
  wenzijieshi varchar(500),
  shengyinjieshi varchar(500),
  wenziliju varchar(500),
  lijulangdu varchar(300),
  status number
)


create  SEQUENCE  vocabulary_sequence
  increment by 1 -- 每次加几个
  start with 1 -- 从1开始计数
  nomaxvalue -- 不设置最大值
  nocycle -- 一直累加，不循环
  cache 10 ;
  
  create or replace trigger auto_id_vocabulary  before insert on vocabulary for each row
when (new.id is null)
begin
     select vocabulary_sequence.nextval into :new.id from dual;
end;



//创建词汇例句表

create table vocabulary_sentences(
    id number NOT NULL PRIMARY KEY,
 	vocabularyid number,
 	lijuwenzi varchar(500),
 	shengyinwenjian varchar(300)
 	
)

create  SEQUENCE  vocabulary_sentences_sequence
  increment by 1 -- 每次加几个
  start with 1 -- 从1开始计数
  nomaxvalue -- 不设置最大值
  nocycle -- 一直累加，不循环
  cache 10 ;
  
  create or replace trigger auto_id_vocabulary_sentences  before insert on vocabulary_sentences for each row
when (new.id is null)
begin
     select vocabulary_sentences_sequence.nextval into :new.id from dual;
end;
