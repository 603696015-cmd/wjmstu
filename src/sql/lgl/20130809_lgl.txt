create table mac_address(
  id int NOT NULL PRIMARY KEY,
  mac varchar(500),
  addtime date
)


create  SEQUENCE  mac_address_sequence
  increment by 1 -- 每次加几个
  start with 1 -- 从1开始计数
  nomaxvalue -- 不设置最大值
  nocycle -- 一直累加，不循环
  cache 10 ;
  
  create or replace trigger auto_id_mac_address  before insert on mac_address for each row
when (new.id is null)
begin
     select mac_address_sequence.nextval into :new.id from dual;
end;
