create or replace procedure intelligent_login_pro(userid_ number,notLogin3day_ number,classid_ number,flag_ out number,returnloginid_ out number)
as
today_ varchar2(50);
count_ number;
loginid_ number;
score_ number;--每次登录得分
maxscore_ number;--总分
begin
     count_:=0;
     loginid_:=0;
     today_ := TO_CHAR(SYSDATE,'YYYYMMDD');--当前日期字符串
     score_:=0.0;
     maxscore_:=0.0;
     
     if notLogin3day_=1 then--3天未登陆
        score_:=-0.1;
        insert into intelligent_login (userid,begintime,today,score,classid) values (userid_,sysdate,today_,score_,classid_);
        flag_:=-1;
     else--不是3天未登录
        select count(1) into count_ from intelligent_login where today=today_ and userid=userid_;
        if count_=0 then--当天第一次登陆
           score_:=0.1;
           insert into intelligent_login (userid,begintime,today,score,classid) values (userid_,sysdate,today_,score_,classid_);
           flag_:=1;
        else--当天不是第一次登陆
           insert into intelligent_login (userid,begintime,today,score,classid) values (userid_,sysdate,today_,score_,classid_);
           flag_:=0;
        end if;
     end if;
     select intelligent_login_sequence.currval into loginid_ from dual;
     returnloginid_:=loginid_;
     
     --判断分数是否已经达到饱和
     select max(score) into maxscore_ from intelligent_login where userid=userid_ and begintime<sysdate;
     if maxscore_ IS NULL then
        maxscore_:=0.0;
     end if;
     maxscore_:=maxscore_+score_;
     if maxscore_>=10  then
        maxscore_:=10.0;
     else if maxscore_<=0 then
        maxscore_:=0.0;  
     end if;
     end if;
     count_:=0;
     select count(1) into count_ from intelligent_login_t where userid=userid_;
     if count_=0 then
        insert into intelligent_login_t (userid,totalscore,classid) values (userid_,maxscore_,classid_);
     else 
        update intelligent_login_t set totalscore=maxscore_ where userid=userid_;
     end if;
     
end;






create or replace procedure intelligent_loginout_pro(userid_ number,loginid_ number)
as
--between_min_ number;
between_sec_ number;
begin
     --between_min_:=0;
     between_sec_:=0;
     update intelligent_login set endtime=sysdate where id=loginid_;--更新退出时间
     --计算登录时间和退出时间差
     select (cast(endtime as date)-cast(begintime as date))*86400 interval into between_sec_ from  intelligent_login  where id=loginid_;
     if between_sec_>0 then
        --between_min_ := round(between_sec_/60,0);
        update intelligent_login set logintime = between_sec_ where id=loginid_;
     end if;
end;
