create or replace procedure sc_cpage (userid_ number, cpid_ number,courseid_ number,classid_ number,passtime_ number)
 AS ispassed_ number ; cp_time_ number ; yxcp_time_ number ;
 process_ number;process__ number;process___ number;scp_getcredit number;
 passed2_ number;passed_ number;scpcnt number;passtime2_ number;islink_ number;incomplete number;exurl_ varchar(4000);initcompliance_ number;
BEGIN
  -- 更新实际时长
  if cpid_>0 then
      select count(cpid) into scpcnt from (select * from study_cpage where userid =userid_ and classid=classid_ ) sc join course_page cp on cp.id = sc.cpid where cp.courseid=courseid_ ;
      if scpcnt>0 then
          --select during,getcredit into cp_time_,scp_getcredit from course_page where id = cpid_;
          update study_cpage set endtime =sysdate,passtime_2 = passtime_2+passtime_,passtime = passtime +passtime_ where userid = userid_ and cpid=cpid_ and classid=classid_ ;       
          sc_cpage_tprocess(  userid_ ,cpid_ ,classid_ );
       end if;
       --process_:=0;
       --select sum(process_t),sum(passtime),sum(passtime_2) into process_,yxcp_time_,passtime2_ from study_cpage scp left join course_page cp on cp.id=scp.cpid where scp.userid = userid_ and scp.classid=classid_ and cp.courseid=courseid_;
       --update study_course set passtime = yxcp_time_,process=process_,passtime_2=passtime2_ where userid = userid_ and courseid=courseid_ and classid =classid_;
       --sc_cpage_course (userid_,courseid_,classid_ );
   else
      select during,islink,exurl into cp_time_,islink_,exurl_ from course where id = courseid_;
      if islink_!=100 then
          select passtime into yxcp_time_ from study_course where userid =  userid_ and courseid= courseid_ and classid=classid_;
          yxcp_time_ := yxcp_time_+passtime_;
          process_ := (yxcp_time_*100)/(60*cp_time_);
          --判断initCompliance是否初始化已经是1了
          select initcompliance into initcompliance_ from study_course where  userid =userid_ and courseid=courseid_ and classid=classid_;
          if initcompliance_=1 then
               update study_course set passtime = yxcp_time_,process= 100,passtime_2 = passtime_2+passtime_,passed=1 where userid =  userid_ and courseid= courseid_ and classid=classid_;
          else
               update study_course set passtime = yxcp_time_,process= process_,passtime_2 = passtime_2+passtime_ where userid =  userid_ and courseid= courseid_ and classid=classid_;
          end if;

          -- 完毕
          if  yxcp_time_>=cp_time_*60 then
              update study_course set passtime = cp_time_*60 ,passed = 1,process = 100 where userid =userid_ and courseid=courseid_ and classid=classid_;
          end if;
      else
          select count(*) into incomplete from (SELECT ci.identifier,ci.type,ci.title,ci.launch,sus.lessonstatus FROM (select * from sc_iteminfo where (type='sco' or type='asset') and CourseID = exurl_) ci
           left join (select * from SC_UserSCOInfo where userid = to_char(userid_) AND CourseID = exurl_  AND CLASSID=to_char(classid_)) sus  on ci.identifier = sus.scoid order by ci.sequence)
           where lessonstatus is null or trim(lessonstatus) in('','incomplete','failed','not attempted') ;
          if incomplete = 0 then
             update study_course set passtime = cp_time_,process= 100,passed=1,passtime_2 = passtime_2+passtime_ where userid =  userid_ and courseid= courseid_ and classid=classid_;
          end if;
      end if;
      s_course_tprocess(userid_,courseid_,classid_);
   end if;
END;



create or replace procedure sc_cpage_setnull
(userid_ number,cpid_ number,courseid_ number,classid_ number)
as
during_ number;
nextcpid_ number;
nextcpisnull_ number;
cpsortid_ number;
count_ number;
roomid_ number;
ispassed_ number;


begin
     --判断当前章节是否存在考场
     select  count(1) into count_
					 from exam_room er 
					 left join study_exampaper se on se.roomid = er.id and se.userid = userid_ 
					 where er.courseid=courseid_  and er.cpid=cpid_  order by er.sortid asc;
     if count_>0 then
        --查找当前章节sortid最大的考场
        select erid into roomid_ from (select  er.id  erid
					 from exam_room er 
					 left join study_exampaper se on se.roomid = er.id and se.userid = userid_ 
					 where er.courseid=courseid_  and er.cpid=cpid_  order by er.sortid desc) where rownum=1;
        if roomid_>0 then
          --判断该考场是否通过
          select ispassed into ispassed_ from study_room where userid=userid_ and roomid=roomid_;
          if ispassed_=1 then
             --查找当前章节的sortid
             select count(1) into count_ from course_page where courseid=courseid_ and id=cpid_;
             if count_>0 then
               select sortid into cpsortid_ from course_page where courseid=courseid_ and id=cpid_;
               --判断是否有下一章节
               select count(1) into count_ from course_page where courseid=courseid_ and sortid=cpsortid_+1;
               if count_=1 then
                  --存在下一章节，查找下一章节的id和isnull
                  select id,isnull,during into nextcpid_,nextcpisnull_,during_ from course_page where courseid=courseid_ and sortid=cpsortid_+1;
                  --如果下一章节的isnull=1
                  if nextcpisnull_=1 then
                     --更新下一章节的学习
                     update study_cpage set passtime=60*during_,passtime_2=60*during_,passed=1,passed2=0,process=50,process_t=12.5
                     where classid=classid_ and userid=userid_ and cpid=nextcpid_;
                  end if;
               end if;
             end if;
          end if;
        end if;
     else
       --查找当前章节的sortid
       select sortid into cpsortid_ from course_page where courseid=courseid_ and id=cpid_;
       --判断是否有下一章节
       select count(1) into count_ from course_page where courseid=courseid_ and sortid=cpsortid_+1;
       if count_=1 then
          --存在下一章节，查找下一章节的id和isnull
          select id,isnull,during into nextcpid_,nextcpisnull_,during_ from course_page where courseid=courseid_ and sortid=cpsortid_+1;
          --如果下一章节的isnull=1
          if nextcpisnull_=1 then
             --更新下一章节的学习
             update study_cpage set passtime=60*during_,passtime_2=60*during_,passed=1,passed2=0,process=50,process_t=12.5
             where classid=classid_ and userid=userid_ and cpid=nextcpid_;
          end if;
       end if;
     end if;
end;




create or replace procedure sc_setmycpage(userid_ number,
                                              roomid_ number)
as

ispassed_ number;--用户是否考试通过
courseid_ number;--章节对应的课程id
course_roomid_ number;--课程对应的结业考场id
classid_ number;--章节对应课程对应的培训班id
begin
  cpid_:=0;
  ispassed_:=0;
  courseid_:=0;
  classid_:=0;
  course_roomid_:=0;
  --1判断考场是否通过
  select ispassed into ispassed_ from study_room where userid=userid_ and roomid=roomid_;
  if ispassed_=1 then
    select cpid into cpid_ from exam_room where id=roomid_;--章节id
    if cpid_!=0 then
      select courseid into courseid_ from course_page where id=cpid_;
      --获取课程结业考场有问题
      --如果一个课程绑定多个培训班和考场的情况
      select classid into classid_ from study_course where courseid=courseid_ and userid=userid_;
      --select id into course_roomid_ from exam_room where courseid=courseid_ and isnormal=1  and (cpid=0 or cpid is null);
  	  --if course_roomid_!=0 then
  	      --select classid into classid_ from class_course where courseid =courseid_ and eroomid=course_roomid_;
  		  --判断该章节对应考场是否都通过
    		  ispassed_:=get_cpage_examrooms_passed(userid_,courseid_,cpid_);
    		  dbms_output.put_line(ispassed_);
          if ispassed_=1 then
              update study_cpage set passed2=1,process=100.00,process_t=50.00 where userid=userid_ and cpid=cpid_;
              if classid_>0 then
        			   sc_cpage_tprocess(userid_,cpid_,classid_);
                 --如果下一个章节为空
                 --设置下一个章节的学习
                 sc_cpage_setnull(userid_,cpid_,courseid_,classid_);
        		  end if;
          else
              update study_cpage set passed2=0 where userid=userid_ and cpid=cpid_;
          end if;
  	  --end if;   
    end if;
  end if;
end;


commit;
