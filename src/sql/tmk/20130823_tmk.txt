create or replace procedure getMyEpBlocksScore(sqid_ number,myscore_ out float)
  as
    score_ float;
    total_ float;
    cursor myepblocks is
           select myscore from study_blocks where sqid=sqid_;

  begin
     score_:=0.00;
     total_:=0.00;
     open myepblocks;
         loop
            fetch myepblocks into score_ ;
            exit when myepblocks%notfound;

            total_ := total_ + score_ ;

         end loop;
     close myepblocks;
     myscore_ := round(total_,1);
  end;
