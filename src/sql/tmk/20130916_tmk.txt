create or replace procedure getNowCourseid
(classid_ number,userid_ number,flag_ out number)
as
clsid_ number;
orderid_ number;
process_ number;
courseid_ number;
roomid_ number;
passed_ number;
count_ number;
cursor studycourses is
       select cc.classid,cc.orderid,sc.process,sc.courseid,er.id from class_course cc 
       right join study_course sc
       on cc.classid=sc.classid  and sc.classid=classid_ and cc.courseid=sc.courseid
       left join exam_room er 
       on cc.eroomid=er.id
       where sc.userid=userid_ and cc.classid=classid_
       order by cc.orderid asc;
begin
        flag_:=0;
        open studycourses;
         loop
            fetch studycourses into clsid_,orderid_,process_,courseid_,roomid_ ;
            exit when studycourses%notfound;
            if process_<100 then
               flag_:= courseid_;
               exit;
            else
               select count(1) into count_ from study_room where userid=userid_ and roomid=roomid_;
               if count_ = 1 then
                  select ispassed into passed_ from study_room where userid=userid_ and roomid=roomid_;
                 --select passed into passed_ from study_course where classid=classid_ and userid=userid_ and courseid=courseid_;
                 if passed_=0 then
                    flag_:= courseid_;
                    exit;
                 else
                    flag_:= courseid_;
                 end if;
               else 
                 flag_:= courseid_;
                 exit;
               end if;
               
            end if;
         end loop;
       close studycourses;
end;
