create or replace procedure sr_setscore(mepid number) as
  roomid_    number;
  epid_      number;
  userid_    number;
  ep_score   number;
  ep_tscore_   number;
  passgrade_ number;
  ispassed_  number default 0;
  blockid_   number;
  random_    number;
  b_rulestr  varchar2(4000);
  qscore_    number;
  btype_     number;
  cursor epblocks is
    /**
    select epb.id,epb.random,epb.rulestring,nvl(sb.myscore,0),epb.type 
    from exampaperblock epb left join study_blocks sb
    on epb.id=sb.blockid and sb.sqid=mepid
    where epb.exampaperid = epid_ ;
    */
    select id, random, rulestring, eachscore, type
      from exampaperblock
     where exampaperid = epid_;
     
begin
  -- 第一步骤 读取考生考卷信息
  select roomid, epid, userid
    into roomid_, epid_, userid_
    from study_quizinfo
   where id = mepid;
  --2 设置试卷各个大题分值
  open epblocks;
  loop
    fetch epblocks
      into blockid_, random_, b_rulestr, qscore_, btype_; --取值
    exit when epblocks%notfound; --当没有记录时退出循环
    sr_blockscore_set(mepid,
                      userid_,
                      blockid_,
                      random_,
                      b_rulestr,
                      qscore_,
                      btype_);
  end loop;
  close epblocks;
  -- 3 得到大题总分值
  select sum(myscore) into ep_score from study_blocks where sqid = mepid;
  
  select ep_tscore into ep_tscore_ from exampaper where id = epid_;
  select rep.passgrade
    into passgrade_
    from study_quizinfo sqi, exam_reps rep
   where rep.epid = sqi.epid
     and rep.roomid = sqi.roomid
     and sqi.id = mepid;
  
  if ((ep_score * 100) / ep_tscore_) >= passgrade_ then
    ispassed_ := 1;
  end if;
  dbms_output.put_line(ep_score);
  dbms_output.put_line(ep_tscore_);
  dbms_output.put_line(ispassed_);
  -- 4 记录试卷分值
  update study_quizinfo
     set myscore  = nvl(ep_score, 0),
         status   = 2,
         ENDTIME  = sysdate,
         ispassed = nvl(ispassed_, 0)
   where id = mepid;
  --5设置试卷总体得分
  sr_setep_score(userid_,roomid_,epid_);
  -- 6 记录考场分值
  sr_seteroom_score(userid_, roomid_);
end;




create or replace procedure sr_questionscore_set_save
( sqid_ number,blockid_ number,qid_ number ,qindex_ number) as
--qindex_ number 
--TMK添加，判断该问题是第一次答还是第二次答
--第一次答对，分数全得，第二次答对分数为设置的大题第二次答对得分
status_ number default 0;score_ number;q_answeri varchar2(4000);q_rulestri varchar2(4000); u_myanswer varchar2(4000);
u_myansweri varchar2(4000);l_idx number ;u_age number;qcid_ number;qctype_ number;qcanswer_ varchar(4000);qscoreper_ number;
qtype_ number;q_rulestr varchar(4000);random_ number;qscore_ number;q_answer varchar(4000);
cursor question_subs is select id,qtype,answer,scoreper from question where parentid = qid_;
begin
      score_:=0;
      l_idx:=1;
      --if qindex_ = 1 then --第一次答对
         select random, rulestring, eachscore, type into random_,q_rulestr,qscore_,qtype_ from exampaperblock where id = blockid_;
     -- else  --第二次答对
         --select random, rulestring, secondscore, type into random_,q_rulestr,qscore_,qtype_ from exampaperblock where id = blockid_;
      --end if;
      if random_=0 then
        select rulestring into q_rulestr from exampaperblockquestion where blockid = blockid_ and questionid = qid_;
      end if;
      select answer,qtype into q_answer,qtype_ from question where id = qid_;
      select myanswer into u_myanswer from study_questions where sqid =sqid_ and blockid=blockid_ and qid= qid_;
      if qtype_ =1 or qtype_ =2 or qtype_ =3 or qtype_  =4 or qtype_ =5 or qtype_=10 or qtype_=15 then
             if q_answer=u_myanswer then
                score_:=qscore_;
                status_:=1;
             else
                status_:=0;
             end if;
       elsif qtype_ = 7 then
             open question_subs;
             loop
                fetch question_subs into qcid_,qctype_,qcanswer_,qscoreper_;
                exit when question_subs%notfound;
                sr_questionsubscore_set( sqid_,blockid_,qcid_,qctype_,qscore_*qscoreper_/100,qcanswer_);
                l_idx:=l_idx+1;
             end loop;
             CLOSE question_subs;
             select sum(sq.myscore) into score_ from study_questions sq left join question q on q.id = sq.qid where q.parentid = qid_ and
                                 sq.sqid =sqid_ and sq.blockid=blockid_;
             status_:=0;
       elsif qtype_ = 8 then
             select nvl( floor(to_char(sysdate,'yyyy' ))-floor(to_char(u.shengri,'yyyy')),-1) into u_age from eluser u join study_quizinfo sqi on sqi.userid=u.id where sqi.id = sqid_;
             if u_age =-1 then
                status_:=-4 ;
                score_:=0;
             else
               if trim(q_rulestr)='' then
                  score_:=0;
                  status_:=-7;
               else
                 u_myansweri := getdzscoreandstatus( u_myanswer ,q_rulestr ,u_age,qscore_);
                 score_:= getsub(u_myansweri,':',1) ;
                 status_:= nvl( getsub(u_myansweri,':',2),0) ;
               end if;
             end if;
       elsif qtype_ = 9 then
            while l_idx <=6 loop
                   q_answeri := trim(getSub(q_answer,'-=SpEl=-',l_idx));
                   u_myansweri := trim(getSub(u_myanswer,'-=SpEl=-',l_idx));
                   q_rulestri := trim(getSub(q_rulestr,'-=SpRule-',l_idx));
                   l_idx:=l_idx+1;
                   if q_answeri = u_myansweri then
                       score_:=score_+to_number( q_rulestri );
                   end if;
             end loop;
             status_:=1;
       elsif qtype_ = 11 or  qtype_ = 6 then
             status_:=-1;
       else
             status_:=-2;
       end if;
       if  qtype_ != 11 and qtype_ != 6 then
       dbms_output.put_line(qscore_);
       update study_questions set myscore = nvl(score_,0) ,status= nvl(status_,0) where sqid =sqid_ and blockid=blockid_ and qid= qid_;
       end if;
       sr_questionscore_block_set(sqid_ ,blockid_ );
end ;
