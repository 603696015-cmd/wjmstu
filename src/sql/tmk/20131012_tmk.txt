create or replace procedure getprecCourseid
(classid_ number,userid_ number,nowcourseid_ number,flag_ out number)
as
clsid_ number;
orderid_ number;
process_ number;
courseid_ number;
roomid_ number;
passed_ number;
count_ number;
cursor studycourses is
       select cc.classid,cc.orderid,sc.process,sc.courseid,er.id from class_course cc 
       right join study_course sc
       on cc.classid=sc.classid  and sc.classid=classid_ and cc.courseid=sc.courseid
       left join exam_room er 
       on cc.eroomid=er.id
       where sc.userid=userid_ and cc.classid=classid_
       order by cc.orderid asc;
begin
        flag_:=0;
        open studycourses;
         loop
            fetch studycourses into clsid_,orderid_,process_,courseid_,roomid_ ;
            exit when studycourses%notfound;
            --找到上一门课程
            if nowcourseid_=courseid_  then
               if orderid_=1 then
                  flag_:=courseid_;
                  exit;
               else
                  select a.courseid into flag_ from (select courseid from class_course where orderid<orderid_ and classid=classid_ order by orderid desc) a where rownum=1;
                  exit;
               end if;
            end if;
         end loop;
       close studycourses;
end;
