create or replace procedure set_epq_totalcount(exampaperid_ number)
--设置试卷总题目数量
as
blockid_ number;
count_ number;
totalcount_ number;
cursor blocks is
   select id from exampaperblock  where exampaperid=exampaperid_;
cursor blockquestionscount is
   select count(1) from exampaperblockquestion where blockid=blockid_;
begin
   totalcount_:=0;
   open blocks;
     loop
        fetch blocks into blockid_ ;
        exit when blocks%notfound;
        if blockid_>0 then
           open blockquestionscount;
             loop
                fetch blockquestionscount into count_ ;
                exit when blockquestionscount%notfound;
                totalcount_:=totalcount_+count_;
             end loop;
           close blockquestionscount;
        end if;
     end loop;
    close blocks;
    if totalcount_>0 then
       update exampaper set questiontotalcount=totalcount_ where id=exampaperid_;
    end if;
end;



create or replace procedure set_exampapers(id_ number)
as
exampaperid_ number;
cursor exampapers is
   select id from exampaper;
begin
   open exampapers;
     loop
        fetch exampapers into exampaperid_ ;
        exit when exampapers%notfound;
        if exampaperid_>0 then
            set_epq_totalcount(exampaperid_);
        end if;
     end loop;
   close exampapers;
end;
