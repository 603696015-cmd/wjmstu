-- Create table
create table LABLETREE
(
  ID          NUMBER not null,
  PARENTID    NUMBER,
  LID         NUMBER,
  RID         NUMBER,
  NAME        VARCHAR2(200) not null,
  DESCRIPTION VARCHAR2(400),
  BH          VARCHAR2(400),
  STATUS      NUMBER default 0 not null
)
tablespace USERS
  pctfree 10
  initrans 1
  maxtrans 255
  storage
  (
    initial 64K
    minextents 1
    maxextents unlimited
  );
-- Create/Recreate primary, unique and foreign key constraints 
alter table LABLETREE
  add constraint LABLETRE_PRIMARY_ID primary key (ID)
  using index 
  tablespace USERS
  pctfree 10
  initrans 2
  maxtrans 255
  storage
  (
    initial 64K
    minextents 1
    maxextents unlimited
  );
  
  
  -- Create sequence 
create sequence LABLETREE_SEQUENCE
minvalue 1
maxvalue 9999999999
start with 10
increment by 1
cache 10;


--单独执行--
create or replace trigger auto_id_labletree  before insert on  labletree  for each row
when (new.id is null)
begin  select labletree_sequence.nextval into :new.id from dual ;   end;
