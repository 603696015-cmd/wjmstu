create or replace procedure intelligent_points(userid_ number,classid_ number,points_ out float)
as
totalpoints_ float;
sql_ varchar2(4000);
begin
     totalpoints_:=0.00f;
     sql_ := 'select totalscore from intelligent_login_t where userid='||userid_||' and classid='||classid_||
             'union all select totalscore from intelligent_week_t_t where userid='||userid_||' and classid='||classid_||
             'union all select totalscore from intelligent_class_t where userid='||userid_||' and classid='||classid_||
             'union all select totalscore from intelligent_proportion_t where userid='||userid_||' and classid='||classid_||
             'union all select totalscore from intelligent_recoding_t where userid='||userid_||' and classid='||classid_||
             'union all select totalscore from intelligent_academic_t_t where userid='||userid_||' and classid='||classid_||
             'union all select totalscore from intelligent_academic_cou_t_t where userid='||userid_||' and classid='||classid_;
     sql_ := 'select sum(totalscore) from ('||sql_||')';
     execute immediate sql_ into totalpoints_;
     points_:=round(totalpoints_,2);
end;






create or replace procedure intelligent_proportion_pro
(userid_ number,            --用户ID
myexampaperid_ number,      --用户试卷ID
exampaperid_ number,        --考卷ID
blockid_ number,            --大题ID
questionid_ number,         --小题ID
classid_ number,            --培训班ID
courseid_ number,           --课程ID
pageid_ number,             --章节ID
roomid_ number)             --章节考场ID
as
count_ number;
totalcount_ number;         --培训班下所有课程和课程的章节对应特殊类型题型的数量
proporcount_ number;        --复听数量
score1_ number;
score2_ number;
totalscore_ number;
ep_ number;
maxcount_ number;           --复听次数
bl_ float;
cursor eps is
      select erps.epid from class_course cc ,exam_room er,exam_reps erps
      where cc.classid=er.classid and cc.courseid=er.courseid and er.id=erps.roomid and cc.classid=classid_
      union all
      select erps.epid from
      (select cc.courseid from class_course cc ,exam_room er,exam_reps erps
      where cc.classid=er.classid and cc.courseid=er.courseid and er.id=erps.roomid and cc.classid=classid_ ) t,course_page cp
      ,exam_room er,exam_reps erps
      where t.courseid=cp.courseid and cp.id=er.cpid and erps.roomid=er.id;

begin
     --插入记录到复听记录表
     count_:=0;
     totalcount_:=0;
     proporcount_:=0;
     totalscore_:=0.0;
     score1_:=0.0;
     score2_:=0.0;
     maxcount_:=0;
     bl_:=0.0f;
     select count(1) into count_ from INTELLIGENT_PROPORTION 
            where userid=userid_ and myexampaperid=myexampaperid_
            and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
            and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
     if count_=0 then
        insert into INTELLIGENT_PROPORTION(userid,myexampaperid,exampaperid,blockid,questionid,
               classid,courseid,pageid,roomid,count,proportime)
            values (userid_,myexampaperid_,exampaperid_,blockid_,questionid_,
                   classid_,courseid_,pageid_,roomid_,1,sysdate);
     else
         update INTELLIGENT_PROPORTION set count=count+1,proportime=sysdate 
            where userid=userid_ and myexampaperid=myexampaperid_
            and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
            and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
     end if;
     --获取培训班下所有课程对应试卷、所有课程的章节对应试卷的所有特殊类型试题总量
     --总题量
     open eps;
         loop
            fetch eps into ep_ ;
            exit when eps%notfound;
            totalcount_:=totalcount_+getquestioncountbyepid(ep_);
         end loop;
     close eps;
     --复听题目数量
     select count(1) into proporcount_ from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_;
     --dbms_output.put_line(proporcount_);
     --dbms_output.put_line(totalcount_);
     score1_:=round(proporcount_/totalcount_ * 10,1);--复听得分
     if score1_>10 then
        score1_:=10.0;
     end if;
     --复听次数
     select sum(count) into maxcount_ from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_;
     bl_:=maxcount_/totalcount_;
     if bl_>3 then
        score2_:=10.0;
     else
        score2_:=round((bl_/3)*10,1);
     end if;
     --更新复听比例总表
     totalscore_:=score1_ + score2_;
     count_:=0;
     select count(1) into count_ from INTELLIGENT_PROPORTION_T where userid=userid_ and classid=classid_;
     if count_=0 then
        insert into INTELLIGENT_PROPORTION_T(userid,totalscore,classid)
               values(userid_,totalscore_,classid_);
     else
         update INTELLIGENT_PROPORTION_T set totalscore=totalscore_ where userid=userid_ and classid=classid_;
     end if;
end;

create or replace procedure intelligent_academic_pro
 (userid_ number,roomid_ number,classid_ number,courseid_ number,pageid_ number,myexampaperid_ number)
 as
 count_ number;
 myscore_ number;
 totalscore_ number;
 avgscore_ number;
 bl_ number;
 countpagequiz_ number;
 begin
      --得到这次考试的成绩
      select myscore into myscore_ from study_quizinfo where id=myexampaperid_ ;
      myscore_:=round(myscore_,2);
      --插入单次章节考试得分表
      insert into intelligent_academic(userid,classid,courseid,pageid,roomid,score,myexampaperid)
                values (userid_,classid_,courseid_,pageid_,roomid_,myscore_,myexampaperid_);
      select round(sum(score),2),round(avg(score),2) into totalscore_,avgscore_ from intelligent_academic 
             where userid=userid_ and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid_;
      --插入章节考试记录表
      count_:=0;
      select count(1) into count_ from intelligent_academic_t where userid=userid_ and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid_;
      if count_=0 then
         insert into intelligent_academic_t(userid,classid,courseid,pageid,roomid,myexampaperid,avgscore,totalscore)
                values (userid_,classid_,courseid_,pageid_,roomid_,myexampaperid_,avgscore_,totalscore_);
      else
          update intelligent_academic_t set totalscore=totalscore_,avgscore=avgscore_ where userid=userid_ and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid_;
      end if;
      --章节考场数
      select count(1) into countpagequiz_ from 
      (
        select erps.epid from
        (select cc.courseid from class_course cc ,exam_room er,exam_reps erps
        where cc.classid=er.classid and cc.courseid=er.courseid and er.id=erps.roomid and cc.classid=classid_ ) t,course_page cp
        ,exam_room er,exam_reps erps
        where t.courseid=cp.courseid and cp.id=er.cpid and erps.roomid=er.id
      );
      --获取培训班下，所有章节考试总成绩
      totalscore_:=0.00;
      select sum(avgscore) into totalscore_ from intelligent_academic_t where userid=userid_ and classid=classid_ ;
      --获取章节考试智能辅导分
      if countpagequiz_>0 then
         bl_:=totalscore_/countpagequiz_ ;
         if bl_>100 then
            bl_:= 100.0;
         end if;
         totalscore_:=round(bl_ * 0.3,2);
      end if;
      
      --插入或更新考试智能辅导分
      count_:=0;
      select count(1) into count_ from intelligent_academic_t_t where userid=userid_ and classid=classid_;
      if count_=0 then
          insert into intelligent_academic_t_t(userid,totalscore,classid)
                values(userid_,totalscore_,classid_);
      else
          update intelligent_academic_t_t set totalscore=totalscore_
                 where userid=userid_ and classid=classid_;
      end if;
 end;

 create or replace procedure intelligent_academic_cou_pro
 (userid_ number,roomid_ number,classid_ number,courseid_ number,myexampaperid_ number,classtype_ number)
 as
 count_ number;
 myscore_ number;
 totalscore_ number;
 avgscore_ number;
 bl_ number;
 countcoursequiz_ number;
 begin
      --得到这次考试的成绩
      select myscore into myscore_ from study_quizinfo where id=myexampaperid_ ;
      myscore_:=round(myscore_,2);
      --插入单次课程考试得分表
      insert into intelligent_academic_course(userid,classid,courseid,roomid,score,myexampaperid)
                values (userid_,classid_,courseid_,roomid_,myscore_,myexampaperid_);
      select round(sum(score),2),round(avg(score),2) into totalscore_,avgscore_ from intelligent_academic_course
             where userid=userid_ and classid=classid_ and courseid=courseid_  and roomid=roomid_;
      --插入课程考试记录表
      count_:=0;
      select count(1) into count_ from intelligent_academic_course_t where userid=userid_ and classid=classid_ and courseid=courseid_  and roomid=roomid_;
      if count_=0 then
         insert into intelligent_academic_course_t(userid,classid,courseid,roomid,myexampaperid,avgscore,totalscore)
                values (userid_,classid_,courseid_,roomid_,myexampaperid_,avgscore_,totalscore_);
      else
          update intelligent_academic_course_t set totalscore=totalscore_,avgscore=avgscore_ where userid=userid_ and classid=classid_ and courseid=courseid_  and roomid=roomid_;
      end if;
      --课程考场数
      select count(1) into countcoursequiz_ from
      (
        select erps.epid from class_course cc ,exam_room er,exam_reps erps
        where cc.classid=er.classid and cc.courseid=er.courseid
        and er.id=erps.roomid and cc.classid=classid_
      );
      --获取培训班下，所有章节考试总成绩
      totalscore_:=0.00;
      select sum(avgscore) into totalscore_ from intelligent_academic_course_t where userid=userid_ and classid=classid_ ;
      --获取章节考试智能辅导分
      if countcoursequiz_>0 then
         bl_:=totalscore_/countcoursequiz_ ;
         if classtype_ = 0 then--1A到3B
            if bl_>100 then
               bl_:= 100.0;
            end if;
            totalscore_:=round(bl_ * 0.3,2);
            if totalscore_>30 then
               totalscore_:=30.0;
            end if;
         else--4A到6B
            if bl_>150 then
               bl_:= 150.0;
            end if;
            totalscore_:=round(bl_ * 0.07,2);
            if totalscore_>10 then
               totalscore_:=10.0;
            end if;
         end if;
      end if;

      --插入或更新考试智能辅导分
      count_:=0;
      select count(1) into count_ from intelligent_academic_cou_t_t where userid=userid_ and classid=classid_;
      if count_=0 then
          insert into intelligent_academic_cou_t_t(userid,totalscore,classid)
                values(userid_,totalscore_,classid_);

create or replace procedure intelligent_academic_cou_pro
 (userid_ number,roomid_ number,classid_ number,courseid_ number,myexampaperid_ number,classtype_ number)
 as
 count_ number;
 myscore_ number;
 totalscore_ number;
 avgscore_ number;
 bl_ number;
 countcoursequiz_ number;
 begin
      --得到这次考试的成绩
      select myscore into myscore_ from study_quizinfo where id=myexampaperid_ ;
      myscore_:=round(myscore_,2);
      --插入单次课程考试得分表
      insert into intelligent_academic_course(userid,classid,courseid,roomid,score,myexampaperid)
                values (userid_,classid_,courseid_,roomid_,myscore_,myexampaperid_);
      select round(sum(score),2),round(avg(score),2) into totalscore_,avgscore_ from intelligent_academic_course
             where userid=userid_ and classid=classid_ and courseid=courseid_  and roomid=roomid_;
      --插入课程考试记录表
      count_:=0;
      select count(1) into count_ from intelligent_academic_course_t where userid=userid_ and classid=classid_ and courseid=courseid_  and roomid=roomid_;
      if count_=0 then
         insert into intelligent_academic_course_t(userid,classid,courseid,roomid,myexampaperid,avgscore,totalscore)
                values (userid_,classid_,courseid_,roomid_,myexampaperid_,avgscore_,totalscore_);
      else
          update intelligent_academic_course_t set totalscore=totalscore_,avgscore=avgscore_ where userid=userid_ and classid=classid_ and courseid=courseid_  and roomid=roomid_;
      end if;
      --课程考场数
      select count(1) into countcoursequiz_ from
      (
        select erps.epid from class_course cc ,exam_room er,exam_reps erps
        where cc.classid=er.classid and cc.courseid=er.courseid
        and er.id=erps.roomid and cc.classid=classid_
      );
      --获取培训班下，所有章节考试总成绩
      totalscore_:=0.00;
      select sum(avgscore) into totalscore_ from intelligent_academic_course_t where userid=userid_ and classid=classid_ ;
      --获取章节考试智能辅导分
      if countcoursequiz_>0 then
         bl_:=totalscore_/countcoursequiz_ ;
         if classtype_ = 0 then--1A到3B
            if bl_>100 then
               bl_:= 100.0;
            end if;
            totalscore_:=round(bl_ * 0.1,2);
            if totalscore_>10 then
               totalscore_:=10.0;
            end if;
         else--4A到6B
            if bl_>150 then
               bl_:= 150.0;
            end if;
            totalscore_:=round(bl_ * 0.07,2);
            if totalscore_>10 then
               totalscore_:=10.0;
            end if;
         end if;
      end if;

      --插入或更新考试智能辅导分
      count_:=0;
      select count(1) into count_ from intelligent_academic_cou_t_t where userid=userid_ and classid=classid_;
      if count_=0 then
          insert into intelligent_academic_cou_t_t(userid,totalscore,classid)
                values(userid_,totalscore_,classid_);
      else
          update intelligent_academic_cou_t_t set totalscore=totalscore_
                 where userid=userid_ and classid=classid_;
      end if;
 end;
 
 
 
 
 
 
  create or replace procedure intelligent_proportion_pro
(userid_ number,            --用户ID

proporcount_ number;        --复听数量

score1_ number;
score2_ number;
totalscore_ number;
ep_ number;
proportcount_ number;           --复听次数
proportprocess_ number;     --复听次数比例
bl_ float;
dingjiroomid_ number;
cursor eps is
      select erps.epid from class_course cc ,exam_room er,exam_reps erps
      where cc.classid=er.classid and cc.courseid=er.courseid and er.id=erps.roomid and cc.classid=classid_
      union all
      select erps.epid from
      (select cc.courseid from class_course cc ,exam_room er,exam_reps erps
      where cc.classid=er.classid and cc.courseid=er.courseid and er.id=erps.roomid and cc.classid=classid_ ) t,course_page cp
      ,exam_room er,exam_reps erps
      where t.courseid=cp.courseid and cp.id=er.cpid and erps.roomid=er.id;

begin
     select roomid into dingjiroomid_ from exam_room_sys where sys=1 ;
     if dingjiroomid_!=roomid_ then 
       --插入记录到复听记录表
       count_:=0;
       totalcount_:=0;
       proporcount_:=0;
       proporprocess_:=0.00f;
       totalscore_:=0.00;
       score1_:=0.00;
       score2_:=0.00;
       proportcount_:=0;
       proportprocess_:=0.00f;
       bl_:=0.00f;
       select count(1) into count_ from INTELLIGENT_PROPORTION 
              where userid=userid_ and myexampaperid=myexampaperid_
              and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
              and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
       if count_=0 then
          insert into INTELLIGENT_PROPORTION(userid,myexampaperid,exampaperid,blockid,questionid,
                 classid,courseid,pageid,roomid,count,proportime)
              values (userid_,myexampaperid_,exampaperid_,blockid_,questionid_,
                     classid_,courseid_,pageid_,roomid_,1,sysdate);
       else
           update INTELLIGENT_PROPORTION set count=count+1,proportime=sysdate 
              where userid=userid_ and myexampaperid=myexampaperid_
              and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
              and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
       end if;
       --获取培训班下所有课程对应试卷、所有课程的章节对应试卷的所有特殊类型试题总量
       --总题量
       open eps;
           loop
              fetch eps into ep_ ;
              exit when eps%notfound;
              totalcount_:=totalcount_+getquestioncountbyepid(ep_);--总题数
           end loop;
       close eps;
       --复听题目数量
       select count(1) into proporcount_ from (select distinct(questionid) from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_);
       proporprocess_:=round(proporcount_/totalcount_,2);--复听题目数量比例
       score1_:=round(proporcount_/totalcount_ * 10,2);--复听得分
       if score1_>10 then
          score1_:=10.0;
       end if;
       --复听次数
       select sum(count) into proportcount_ from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_;
       proportprocess_:=round(proportcount_/totalcount_,2);
       bl_:=proportcount_/totalcount_;
       if bl_>3 then
          score2_:=10.0;
       else
          score2_:=round((bl_/3)*10,2);
       end if;
       --更新复听比例总表
       totalscore_:=score1_ + score2_;
       count_:=0;
       select count(1) into count_ from INTELLIGENT_PROPORTION_T where userid=userid_ and classid=classid_;
       if count_=0 then
          insert into INTELLIGENT_PROPORTION_T(userid,totalscore,classid,
                 totalcount,Proportionqcount,proportionqprocess,Proportiontcount,proportiontprocess,qscore,tscore)
                 values(userid_,totalscore_,classid_,
                 totalcount_,proporcount_,proporprocess_,proportcount_,proportprocess_,score1_,score2_);
       else
           update INTELLIGENT_PROPORTION_T 
                  set totalscore=totalscore_ ,
                  totalcount = totalcount_,Proportionqcount=proporcount_,
                  proportionqprocess=proporprocess_,
                  Proportiontcount=proportcount_,
                  proportiontprocess=proportprocess_ ,
                  qscore=score1_,
                  tscore=score2_
                  where userid=userid_ and classid=classid_;
       end if;
     end if;
end;

 create or replace procedure intelligent_proportion_pro
(userid_ number,            --用户ID

scoreproportionprocessper_ float,

scoreproportiontimeper_ float
)
as
count_ number;
totalcount_ number;         --培训班下所有课程和课程的章节对应特殊类型题型的数量
proporcount_ number;        --复听数量

score1_ number;
score2_ number;
totalscore_ number;
ep_ number;
proportcount_ number;           --复听次数
proportprocess_ number;     --复听次数比例
bl_ float;
dingjiroomid_ number;
cursor eps is
      select erps.epid from class_course cc ,exam_room er,exam_reps erps
      where cc.classid=er.classid and cc.courseid=er.courseid and er.id=erps.roomid and cc.classid=classid_
      union all
      select erps.epid from
      (select cc.courseid from class_course cc ,exam_room er,exam_reps erps
      where cc.classid=er.classid and cc.courseid=er.courseid and er.id=erps.roomid and cc.classid=classid_ ) t,course_page cp
      ,exam_room er,exam_reps erps
      where t.courseid=cp.courseid and cp.id=er.cpid and erps.roomid=er.id;

begin
     select roomid into dingjiroomid_ from exam_room_sys where sys=1 ;
     if dingjiroomid_!=roomid_ then 
       --插入记录到复听记录表
       count_:=0;
       totalcount_:=0;
       proporcount_:=0;
       proporprocess_:=0.00f;
       totalscore_:=0.00;
       score1_:=0.00;
       score2_:=0.00;
       proportcount_:=0;
       proportprocess_:=0.00f;
       bl_:=0.00f;
       select count(1) into count_ from INTELLIGENT_PROPORTION 
              where userid=userid_ and myexampaperid=myexampaperid_
              and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
              and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
       if count_=0 then
          insert into INTELLIGENT_PROPORTION(userid,myexampaperid,exampaperid,blockid,questionid,
                 classid,courseid,pageid,roomid,count,proportime)
              values (userid_,myexampaperid_,exampaperid_,blockid_,questionid_,
                     classid_,courseid_,pageid_,roomid_,1,sysdate);
       else
           update INTELLIGENT_PROPORTION set count=count+1,proportime=sysdate 
              where userid=userid_ and myexampaperid=myexampaperid_
              and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
              and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
       end if;
       --获取培训班下所有课程对应试卷、所有课程的章节对应试卷的所有特殊类型试题总量
       --总题量
       open eps;
           loop
              fetch eps into ep_ ;
              exit when eps%notfound;
              totalcount_:=totalcount_+getquestioncountbyepid(ep_);--总题数
           end loop;
       close eps;
       --复听题目数量
       select count(1) into proporcount_ from (select distinct(questionid) from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_);
       proporprocess_:=round(proporcount_/totalcount_,2);--复听题目数量比例
       score1_:=round(proporcount_/totalcount_ * scoreproportionprocessper_ * 100,2);--复听得分
       if score1_>round(scoreproportionprocess_,2) then
          score1_:=round(scoreproportionprocess_,2);
       end if;
       --复听次数
       select sum(count) into proportcount_ from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_;
       proportprocess_:=round(proportcount_/totalcount_,2);
       bl_:=proportcount_/totalcount_;
       if bl_>3 then
          score2_:=round(scoreproportiontime_,2);
       else
          score2_:=round((bl_/3)* 100 * scoreproportiontimeper_,2);
          if score2_>round(scoreproportiontime_,2) then
             score2_ := round(scoreproportiontime_,2);
          end if;
       end if;
       --更新复听比例总表
       totalscore_:=score1_ + score2_;
       count_:=0;
       select count(1) into count_ from INTELLIGENT_PROPORTION_T where userid=userid_ and classid=classid_;
       if count_=0 then
          insert into INTELLIGENT_PROPORTION_T(userid,totalscore,classid,
                 totalcount,Proportionqcount,proportionqprocess,Proportiontcount,proportiontprocess,qscore,tscore)
                 values(userid_,totalscore_,classid_,
                 totalcount_,proporcount_,proporprocess_,proportcount_,proportprocess_,score1_,score2_);
       else
           update INTELLIGENT_PROPORTION_T 
                  set totalscore=totalscore_ ,
                  totalcount = totalcount_,Proportionqcount=proporcount_,
                  proportionqprocess=proporprocess_,
                  Proportiontcount=proportcount_,
                  proportiontprocess=proportprocess_ ,
                  qscore=score1_,
                  tscore=score2_
                  where userid=userid_ and classid=classid_;
       end if;
     end if;
end;

      where t.courseid=cp.courseid and cp.id=er.cpid and erps.roomid=er.id;

begin
     select roomid into dingjiroomid_ from exam_room_sys where sys=1 ;
     if dingjiroomid_!=roomid_ then 
       --插入记录到复听记录表
       count_:=0;
       totalcount_:=0;
       proporcount_:=0;
       proporprocess_:=0.00f;
       totalscore_:=0.00;
       score1_:=0.00;
       score2_:=0.00;
       proportcount_:=0;
       proportprocess_:=0.00f;
       bl_:=0.00f;
       select count(1) into count_ from INTELLIGENT_PROPORTION 
              where userid=userid_ and myexampaperid=myexampaperid_
              and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
              and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
       if count_=0 then
          insert into INTELLIGENT_PROPORTION(userid,myexampaperid,exampaperid,blockid,questionid,
                 classid,courseid,pageid,roomid,count,proportime)
              values (userid_,myexampaperid_,exampaperid_,blockid_,questionid_,
                     classid_,courseid_,pageid_,roomid_,1,sysdate);
       else
           update INTELLIGENT_PROPORTION set count=count+1,proportime=sysdate 
              where userid=userid_ and myexampaperid=myexampaperid_
              and exampaperid=exampaperid_ and blockid=blockid_ and questionid=questionid_
              and classid=classid_ and courseid=courseid_ and pageid=pageid_ and roomid=roomid;
       end if;
       --获取培训班下所有课程对应试卷、所有课程的章节对应试卷的所有特殊类型试题总量
       --总题量
       open eps;
           loop
              fetch eps into ep_ ;
              exit when eps%notfound;
              totalcount_:=totalcount_+getquestioncountbyepid(ep_);--总题数
           end loop;
       close eps;
       --复听题目数量
       select count(1) into proporcount_ from (select distinct(questionid) from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_);
       proporprocess_:=round(proporcount_/totalcount_,2);--复听题目数量比例
       score1_:=round(proporcount_/totalcount_ * scoreproportionprocessper_ * 100,2);--复听得分
       if score1_>round(scoreproportionprocess_,2) then
          score1_:=round(scoreproportionprocess_,2);
       end if;
       --复听次数
       select sum(count) into proportcount_ from INTELLIGENT_PROPORTION where userid=userid_ and classid=classid_;
       proportprocess_:=round(proportcount_/totalcount_,2);
       bl_:=proportcount_/totalcount_;
       if bl_>=3 then
          score2_:=round(scoreproportiontime_,2);
       else
          score2_:=round(bl_* 100 * scoreproportiontimeper_,2);
          if score2_>round(scoreproportiontime_,2) then
             score2_ := round(scoreproportiontime_,2);
          end if;
       end if;
       --更新复听比例总表
       totalscore_:=score1_ + score2_;
       count_:=0;
       select count(1) into count_ from INTELLIGENT_PROPORTION_T where userid=userid_ and classid=classid_;
       if count_=0 then
          insert into INTELLIGENT_PROPORTION_T(userid,totalscore,classid,
                 totalcount,Proportionqcount,proportionqprocess,Proportiontcount,proportiontprocess,qscore,tscore)
                 values(userid_,totalscore_,classid_,
                 totalcount_,proporcount_,proporprocess_,proportcount_,proportprocess_,score1_,score2_);
       else
           update INTELLIGENT_PROPORTION_T 
                  set totalscore=totalscore_ ,
                  totalcount = totalcount_,Proportionqcount=proporcount_,
                  proportionqprocess=proporprocess_,
                  Proportiontcount=proportcount_,
                  proportiontprocess=proportprocess_ ,
                  qscore=score1_,
                  tscore=score2_
                  where userid=userid_ and classid=classid_;
       end if;
     end if;
end;
