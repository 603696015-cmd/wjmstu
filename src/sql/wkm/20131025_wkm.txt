create or replace procedure sr_setscore_wsj(mepid number,classid_ number) as
  roomid_    number;
  epid_      number;
  userid_    number;
  ep_score   number;
  ep_tscore_   number;
  passgrade_ number;
  ispassed_  number default 0;
  blockid_   number;
  random_    number;
  b_rulestr  varchar2(4000);
  qscore_    number;
  btype_     number;
  cursor epblocks is
    /**
    select epb.id,epb.random,epb.rulestring,nvl(sb.myscore,0),epb.type
    from exampaperblock epb left join study_blocks sb
    on epb.id=sb.blockid and sb.sqid=mepid
    where epb.exampaperid = epid_ ;
    */
    select id, random, rulestring, eachscore, type
      from exampaperblock
     where exampaperid = epid_;

begin
  -- 第一步骤 读取考生考卷信息
  select roomid, epid, userid
    into roomid_, epid_, userid_
    from study_quizinfo
   where id = mepid;
  --2 设置试卷各个大题分值
  open epblocks;
  loop
    fetch epblocks
      into blockid_, random_, b_rulestr, qscore_, btype_; --取值
    exit when epblocks%notfound; --当没有记录时退出循环
    sr_blockscore_set(mepid,
                      userid_,
                      blockid_,
                      random_,
                      b_rulestr,
                      qscore_,
                      btype_);
  end loop;
  close epblocks;
  -- 3 得到大题总分值
  select sum(myscore) into ep_score from study_blocks where sqid = mepid;

  select ep_tscore into ep_tscore_ from exampaper where id = epid_;
  select rep.passgrade
    into passgrade_
    from study_quizinfo sqi, exam_reps rep
   where rep.epid = sqi.epid
     and rep.roomid = sqi.roomid
     and sqi.id = mepid;

  if ((ep_score * 100) / ep_tscore_) >= passgrade_ then
    ispassed_ := 1;
  end if;
  dbms_output.put_line(ep_score);
  dbms_output.put_line(ep_tscore_);
  dbms_output.put_line(ispassed_);
  -- 4 记录试卷分值
  update study_quizinfo
     set myscore  = nvl(ep_score, 0),
         status   = 2,
         ENDTIME  = sysdate,
         ispassed = nvl(ispassed_, 0)
   where id = mepid;
  --5设置试卷总体得分
  sr_setep_score_wsj(userid_,roomid_,epid_,classid_);
  -- 6 记录考场分值
  sr_seteroom_score_wsj(userid_, roomid_,classid_);
end;





create or replace procedure sr_setep_score_wsj(userid_ number,
                                           roomid_ number,
                                           epid_   number,
                                           classid_ number ) as
  ispassed_   number;
  maxscore_   number;
  avgscore_   number;
  quizcount_  number;
  myscore_    number;
  passmanner_ number;
begin
  SELECT max(sqi.ispassed),
         max(sqi.myscore),
         avg(sqi.myscore),
         count(sqi.id)
    into ispassed_, maxscore_, avgscore_, quizcount_
    FROM study_quizinfo sqi
   where sqi.userid = userid_
     and sqi.roomid = roomid_
     and sqi.epid = epid_  and sqi.classid=classid_ ;
  select passmanner
    into passmanner_
    from exam_reps
   where epid = epid_
     and roomid = roomid_;
  myscore_ := maxscore_;
  if passmanner_ = 1 then
    myscore_ := avgscore_;
  end if;
  update study_exampaper sep
     set ispassed  = ispassed_,
         maxscore  = nvl(maxscore_,0),
         avgscore  = nvl(avgscore_,0),
         quizcount = nvl(quizcount_,0),
         myscore   = nvl(myscore_,0),
         status = 3
   where sep.userid = userid_
     and sep.roomid = roomid_
     and sep.epid = epid_
     and sep.classid = classid_ ;
end;
