6.创建培训批次表
create table batch
(
  id          NUMBER not null,
  name        VARCHAR2(500) not null,
  description VARCHAR2(4000)
 
);

create  SEQUENCE batch_sequence
	increment by 1 -- 每次加几个
	start with 1 -- 从1开始计数
	nomaxvalue -- 不设置最大值
	nocycle -- 一直累加，不循环
	cache 10 ;
create trigger auto_id_batch before insert on batch for each row 
when (new.id is null)
begin
     select batch_sequence.nextval into :new.id from dual;
end;
